# Start Server

Start the ActualChat App.Server in the background with multi-worktree support.

## Configuration

Server reads `.env` file from project root (auto-generated by `c.ps1` for worktrees).

## Bash Implementation

```bash
#!/bin/bash
set -e

PROJECT_PATH="${AC_ProjectPath:-$(pwd)}"
WT="${AC_Worktree:-}"

# Read .env to extract port and instance for display/PID file naming
# (The app loads .env automatically via DotNetEnv)
if [ -f "$PROJECT_PATH/.env" ]; then
    PORT=$(grep -E '^urls=' "$PROJECT_PATH/.env" | grep -oE '[0-9]+$' || echo "7080")
    INSTANCE=$(grep -E '^CoreSettings__Instance=' "$PROJECT_PATH/.env" | cut -d= -f2 || echo "dev")
else
    PORT="7080"
    INSTANCE="dev"
fi

# Determine tmp directory
if [ "$AC_OS" = "Linux in Docker" ]; then
    TMP_DIR="/tmp"
else
    TMP_DIR="$PROJECT_PATH/tmp"
fi
mkdir -p "$TMP_DIR"

PID_FILE="$TMP_DIR/server-$INSTANCE.pid"
LOG_FILE="$TMP_DIR/server-$INSTANCE.log"

echo "Starting server: $INSTANCE on port $PORT"

# Check if already running
if [ -f "$PID_FILE" ]; then
    EXISTING_PID=$(cat "$PID_FILE")
    if kill -0 "$EXISTING_PID" 2>/dev/null; then
        echo "Already running (PID: $EXISTING_PID)"
        echo "URL: http://localhost:$PORT"
        exit 0
    fi
    rm -f "$PID_FILE"
fi

# Check port availability
if lsof -i :"$PORT" 2>/dev/null | grep -q LISTEN; then
    echo "Error: Port $PORT already in use" >&2
    exit 1
fi

# Start server (App.Server loads .env automatically via DotNetEnv)
export ActualChat_CaptchaBypassEnabled=true

SERVER_PROJECT="$PROJECT_PATH/src/dotnet/App.Server/App.Server.csproj"

# Always use Development environment for local servers
export ASPNETCORE_ENVIRONMENT=Development

# App.Server loads .env from solution root automatically, --no-launch-profile avoids launchSettings.json overrides
dotnet run --project "$SERVER_PROJECT" --no-launch-profile > "$LOG_FILE" 2>&1 &

echo $! > "$PID_FILE"

echo "Started (PID: $(cat $PID_FILE))"
echo "Log: $LOG_FILE"
echo "URL: http://localhost:$PORT"
if [ -n "$WT" ]; then
    echo "Browser: https://$WT.local.voxt.ai"
else
    echo "Browser: https://local.voxt.ai"
fi
```
